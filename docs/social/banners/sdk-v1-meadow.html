import { verifyReceipts, parseReceiptInput,
  createRpcChainReader, ARC_REGISTRIES, arc } from "@arcveil/sdk"

const chain = createRpcChainReader({ chainId: arc.id,
  endpoint: arc.rpcUrls.default.http[0], ...ARC_REGISTRIES[arc.id] })

const parsed = parseReceiptInput(json)
const report = await verifyReceipts(parsed.receipts, { chain })
